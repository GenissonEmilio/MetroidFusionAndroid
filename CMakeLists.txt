cmake_minimum_required(VERSION 3.20)
project(FusionRecomp C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ANDROID)
    include(cmake/Android.cmake)
    return()
endif()
set(GBARECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/gbarecomp" CACHE PATH "Shared GBA core")
set(GBARECOMP_GENERATED_BIOS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated/open_bios" CACHE PATH "Open firmware corpus" FORCE)
set(GBARECOMP_ENABLE_MODS ON CACHE BOOL "Built-in album hooks" FORCE)
add_subdirectory("${GBARECOMP_ROOT}" gbarecomp_build EXCLUDE_FROM_ALL)
if(MSVC AND NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(gbarecomp_runtime PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/src/msvc_compat.h")
endif()
file(GLOB FUSION_GENERATED CONFIGURE_DEPENDS "generated/*.cpp")
add_executable(FusionRecomp src/main.cpp ${FUSION_GENERATED})
target_include_directories(FusionRecomp PRIVATE src "${GBARECOMP_ROOT}/src/runtime" "${GBARECOMP_ROOT}/src/armv4t" "${GBARECOMP_ROOT}/src/gba")
if(MSVC)
    target_link_libraries(FusionRecomp PRIVATE gbarecomp_runtime gbarecomp_debug gbarecomp_gba gbarecomp_armv4t)
    target_link_options(FusionRecomp PRIVATE /STACK:16777216)
else()
    target_link_libraries(FusionRecomp PRIVATE "-Wl,--start-group" gbarecomp_runtime gbarecomp_debug gbarecomp_gba gbarecomp_armv4t "-Wl,--end-group")
endif()
target_compile_definitions(FusionRecomp PRIVATE GBARECOMP_DEFAULT_DEBUG_PORT=19844 GBARECOMP_WINDOW_TITLE="Metroid Fusion" GBARECOMP_DEFAULT_GAME_CONFIG="game.toml")
